SG Wayfinder / scoring.jsView on GitHub →

How a guess is scored from its distance to the target. Excerpt of the real file.

// Scoring. Three parts per round, 1,000 points each.

export const MAX_PER_PART = 1000;
export const MAX_PER_ROUND = MAX_PER_PART * 3;

/** Location: full marks within 75 m, fading with distance (1 km off is about 730). */
export function orientationScore(distanceM) {
  const d = Math.max(0, distanceM - 75);
  return Math.round(MAX_PER_PART * Math.exp(-d / 3000));
}

/** Prediction: relative error against the model time for the player's own route. */
export function predictionScore(predictedMin, actualMin) {
  if (!Number.isFinite(predictedMin) || predictedMin <= 0) return 0;
  const err = Math.abs(predictedMin - actualMin) / Math.max(actualMin, 5);
  return Math.round(MAX_PER_PART * Math.exp(-3.2 * err));
}

/** Plan: how close the whole journey (time plus money) is to the best-value option. */
export function planScore(referenceValue, actualValue) {
  if (!(actualValue > 0)) return 0;
  const ratio = Math.min(1, referenceValue / actualValue);
  return Math.round(MAX_PER_PART * ratio ** 2);
}

A seeded random number generator, so every round is reproducible. Excerpt of the real file.

function xmur3(str) {
  let h = 1779033703 ^ str.length;
  for (let i = 0; i < str.length; i++) {
    h = Math.imul(h ^ str.charCodeAt(i), 3432918353);
    h = (h << 13) | (h >>> 19);
  }
  return () => {
    h = Math.imul(h ^ (h >>> 16), 2246822507);
    h = Math.imul(h ^ (h >>> 13), 3266489909);
    return (h ^= h >>> 16) >>> 0;
  };
}

function mulberry32(a) {
  return () => {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export function createRng(seed) {
  const next = mulberry32(xmur3(String(seed))());
  return {
    next,
    float: (min, max) => min + next() * (max - min),
    /** Integer in [min, max] inclusive. */
    int: (min, max) => min + Math.floor(next() * (max - min + 1)),
    // pick() and shuffle() omitted here
  };
}

Great-circle distance between two points on the Earth. Excerpt of the real file.

const EARTH_R = 6371008.8; // metres
const rad = (d) => (d * Math.PI) / 180;
const deg = (r) => (r * 180) / Math.PI;

/** Great-circle distance in metres. */
export function haversine(a, b) {
  const dLat = rad(b.lat - a.lat);
  const dLng = rad(b.lng - a.lng);
  const s =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(rad(a.lat)) * Math.cos(rad(b.lat)) * Math.sin(dLng / 2) ** 2;
  return 2 * EARTH_R * Math.asin(Math.min(1, Math.sqrt(s)));
}

/** Initial bearing from a to b in degrees, 0-360. */
export function bearing(a, b) {
  const y = Math.sin(rad(b.lng - a.lng)) * Math.cos(rad(b.lat));
  const x =
    Math.cos(rad(a.lat)) * Math.sin(rad(b.lat)) -
    Math.sin(rad(a.lat)) * Math.cos(rad(b.lat)) * Math.cos(rad(b.lng - a.lng));
  return (deg(Math.atan2(y, x)) + 360) % 360;
}

// Clearway Lunar — analytic ephemeris and lunar orientation.
//
// Sun:  Meeus, Astronomical Algorithms ch. 25 (≈0.01°), with annual aberration.
// Moon: Meeus ch. 47, the truncated ELP-2000/82 lunar theory (≈10" in longitude).
// Moon orientation: IAU WGCCRE 2009 model (Archinal et al. 2011), which approximates the
//       Mean Earth / polar axis frame used by LOLA maps and JPL Horizons lunar sites.
// Positions are geometric, in km, in the J2000 equatorial frame unless noted.
// Validated against JPL Horizons (DE441) in tests/lunar.test.mjs.

export const DEG = Math.PI / 180;
export const R_MOON = 1737.4;          // km, IAU mean radius (also LOLA reference sphere)
const TT_MINUS_UTC = 37 + 32.184;      // s (leap seconds since 2017-01-01 + 32.184 s)

/** Julian centuries (TT ≈ TDB) since J2000.0 for a UTC timestamp in ms. */
export function centuriesTT(tMs) {
  const jdTT = (tMs + TT_MINUS_UTC * 1000) / 86400000 + 2440587.5;
  return (jdTT - 2451545.0) / 36525;
}

/** Mean obliquity of the ecliptic of date (rad). */
export function obliquity(T) {
  return (23.439291111 - (46.815 * T + 0.00059 * T * T - 0.001813 * T * T * T) / 3600) * DEG;
}

      var COMMON = ['password', 'qwerty', 'letmein', 'welcome', 'admin', 'iloveyou', 'monkey', 'dragon', 'sherlock', 'singapore'];
  function strength(pw) {
    var pool = 0;
    if (/[a-z]/.test(pw)) pool += 26;
    if (/[A-Z]/.test(pw)) pool += 26;
    if (/[0-9]/.test(pw)) pool += 10;
    if (/[^a-zA-Z0-9]/.test(pw)) pool += 33;
    var bits = pw.length && pool ? pw.length * Math.log2(pool) : 0;
    var low = pw.toLowerCase();
    var common = COMMON.some(function (w) { return low.indexOf(w) > -1; });
    var seq = /(abc|bcd|cde|123|234|345|456|567|678|789|qwe|wer|ert)/i.test(pw);
    var rep = /(.)\1\1/.test(pw);
    if (common) bits = Math.min(bits, 20);
    if (seq) bits -= 8;
    if (rep) bits -= 8;
    bits = Math.max(0, bits);
    var secs = Math.pow(2, bits) / 1e10;
    return { bits: bits, secs: secs, common: common, seq: seq, rep: rep, pool: pool };
  }

    Security tool · Infocomm Club
    The idea
    Built for the Infocomm Club's Discovery Fortnight session, where students write their own password checker in Python. Students enter details a stranger could read off social media, then a password, and see how quickly it would fall.
    What it shows
    A live entropy meter with the effective value (what a smart attacker needs) beside the theoretical one (from length and character types alone). The gap between the two is the lesson. It also gives crack times under four attacks, takes the password apart into the pieces an attacker prices, and generates a stronger one.
    How it estimates
    It prices every way of splitting the password into pieces and keeps the cheapest, in the style of zxcvbn: common-password ranks, a dictionary of about 57,000 English words, your own details, dates, sequences and keyboard walks. It is a set of explainable rules, not machine learning, and the project says its estimates are rough.
    Privacy
    Everything runs in the browser. Nothing is sent anywhere or stored. Generated passwords use the browser's secure random generator.

    20 automated tests cover the engine. It runs with no build step.

    Remote control · private repo
    The idea
    Use an iPad as a window onto my Windows laptop without installing an app. The laptop runs a small Python server, and the iPad just opens a page in Safari.
    Two modes
    Tabs extends my Chrome tabs onto the iPad so I can use them there. Screen mirrors the whole laptop screen and lets me control it by touch, with taps, drags, scrolling and typing.
    Security first
    The connection uses TLS with a certificate made on the laptop. The iPad pairs by typing a short code that is never sent over the network (SRP), a second layer of AES-256-GCM encryption sits on top, and the laptop asks me to approve every new device. Codes are single use and expire, wrong guesses are limited, messages are strictly checked, and nothing I do is logged.
    Speed and quality
    Screen mode sends only the parts of the picture that changed, captures and encodes at the same time, slows down when the iPad falls behind instead of building up delay, and sends a sharper frame when the screen goes still.
    Status
    It works end to end on my own laptop and iPad, and I am still tuning speed and sharpness. It has 56 automated tests, including ones that drive real browser pages. The repository is private while I test it, so there is no public code link.

    Security training kit · open source
    The idea
    A free practice kit for capture-the-flag (CTF) competitions, made for the Infocomm Club after placing third at Cyber Sports Day. Everything runs in the browser, so a student can start with nothing to install and no account.
    What is in it
    26 puzzles worth 4,950 points, from 50 to 350 points each. Ciphers: Atbash, binary, Morse, Caesar, layered encodings, Vigenère, rail fence, XOR, substitution and affine. Hidden in images: file metadata, low-contrast text, data after the end of a file, bit planes and least-significant-bit messages. Web flaws: robots.txt, view source, URL and cookie tampering, IDOR, path traversal, SQL injection, command injection, reflected XSS and a forged login token.
    Tools and scoring
    A cipher workbench and an image lab sit beside the puzzles, so players learn how the techniques work, not only the answer. Hints cost points, and a result code lets a club compare scores on a leaderboard.
    Safe by design
    The vulnerable websites are small simulations running in a sealed sandbox, so nothing real is attacked and no puzzle can touch the player's browser. Flags are stored only as hashes, and progress stays in the player's own browser.
    My role
    Designed and built it, including writing every puzzle.
    • Each category is ordered from easy to hard, so a beginner and an experienced player can both use it.
    • The simulated browser has a page view, a source view and an editable cookie jar, and the SQL injection puzzle runs a small SQL parser, so it behaves like a real login query.
    • More than 50 automated tests solve every puzzle the way a player would, and a real browser plays through the whole kit.
